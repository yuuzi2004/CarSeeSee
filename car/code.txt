# 验证GPU是否可用（关键！确保用GPU训练）
import torch
print(f"GPU是否可用: {torch.cuda.is_available()}")

print(f"当前GPU: {torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'CPU'}")

# 导入YOLO核心库
from ultralytics import YOLO
import os
import yaml

# 定义数据集根路径（务必与你的实际路径一致）
dataset_root = "/home_data/hejx/yolo_driving_dataset"

# 校验关键文件/文件夹是否存在
check_paths = [
    dataset_root,
    os.path.join(dataset_root, "images/train"),
    os.path.join(dataset_root, "images/val"),
    os.path.join(dataset_root, "labels/train"),
    os.path.join(dataset_root, "labels/val"),
    os.path.join(dataset_root, "driving_data_gpu.yaml")
]

for path in check_paths:
    if os.path.exists(path):
        print(f"✅ {path} 存在")
    else:
        print(f"❌ {path} 不存在，请检查路径！")

# 读取并验证yaml配置
with open(os.path.join(dataset_root, "driving_data_gpu.yaml"), 'r', encoding='utf-8') as f:
    yaml_data = yaml.safe_load(f)
print("📄 数据集配置内容：")
for k, v in yaml_data.items():
    print(f"  {k}: {v}")

# 修正yaml中的绝对路径（避免相对路径问题）
yaml_data['path'] = dataset_root  # 强制指定数据集根目录
# 重新保存修正后的yaml（可选，也可直接用修正后的字典）
with open(os.path.join(dataset_root, "driving_data_gpu_fixed.yaml"), 'w', encoding='utf-8') as f:
    yaml.dump(yaml_data, f, indent=2)
print(f"\n✅ 修正后的配置文件已保存至: {os.path.join(dataset_root, 'driving_data_gpu_fixed.yaml')}")

# 1. 加载预训练YOLOv8模型（可选n/s/m/l/x版本，n最快，x精度最高）
# 推荐先从yolov8s开始，平衡速度和精度
model = YOLO("yolov8s.pt")  # 自动下载预训练权重

# 2. 定义训练参数（GPU训练核心配置）
train_params = {
    "data": os.path.join(dataset_root, "driving_data_gpu_fixed.yaml"),  # 数据集配置文件
    "epochs": 50,  # 训练轮数（可根据效果调整，建议先跑50轮看收敛）
    "batch": 8,  # 批次大小（根据GPU显存调整：12G显存建议16，24G建议32）
    "imgsz": 320,  # 输入图片尺寸（YOLO默认640，可改800/1024提升精度）
    "device": "cuda",  # 指定GPU编号（0表示第一个GPU，多GPU写[0,1]）
    "workers": 4,  # 数据加载线程数（根据CPU核心数调整）
    "lr0": 0.001,  # 初始学习率
    "lrf": 0.1,  # 最终学习率因子
    "weight_decay": 0.0005,  # 权重衰减
    "momentum": 0.937,  # 动量
    "patience": 10,  # 早停耐心值（50轮没提升就停止）
    "save": True,  # 保存最佳模型
    "project": "driving_safety_train",  # 训练结果保存目录
    "name": "yolov8s_driving",  # 实验名称
    "exist_ok": True,  # 覆盖已有实验目录
    "pretrained": True,  # 使用预训练权重
    "optimizer": "SGD",  # 优化器（可选SGD/Adam/AdamW）
    "cos_lr": True,  # 使用余弦学习率衰减
    "val": True,  # 训练时同步验证
    "augment": True,  # 启用数据增强
    "hsv_h": 0.05,  # 色调增强
    "hsv_s": 0.7,    # 饱和度增强
    "hsv_v": 0.5,    # 明度增强
    "degrees": 0.0,  # 旋转角度
    "translate": 0.1, # 平移
    "scale": 0.5,    # 缩放
    "shear": 0.0,    # 剪切
    "perspective": 0.0, # 透视变换
    "flipud": 0.0,   # 上下翻转概率
    "fliplr": 0.5,   # 左右翻转概率
    "mosaic": 1.0,   # Mosaic增强概率
    "mixup": 0.2,    # Mixup增强概率
}

# 3. 开始训练（自动使用GPU）
results = model.train(**train_params)

# 4. 训练完成后，验证模型性能
val_results = model.val()
print(f"✅ 验证集mAP50: {val_results.box.map50:.4f}")
print(f"✅ 验证集mAP50-95: {val_results.box.map:.4f}")

# 5. 保存最终模型（可选，训练后会自动保存）
model.save(os.path.join(train_params['project'], train_params['name'], "final_driving_model.pt"))
print(f"✅ 最终模型已保存至: {os.path.join(train_params['project'], train_params['name'], 'final_driving_model.pt')}")


模型推理验证：（验证训练效果）
# 随机选择一张验证集图片测试
import random
import cv2
import matplotlib.pyplot as plt

# 获取验证集图片列表
val_img_paths = [os.path.join(dataset_root, "images/val", f) for f in os.listdir(os.path.join(dataset_root, "images/val")) if f.endswith(('.jpg', '.png', '.jpeg'))]
test_img_path = random.choice(val_img_paths)

# 模型推理
results = model(test_img_path)

# 可视化结果
img = cv2.imread(test_img_path)
img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
plt.figure(figsize=(12, 8))
plt.imshow(results[0].plot()[:, :, ::-1])  # plot()返回带标注的图片
plt.axis('off')
plt.title(f"Test Result: {os.path.basename(test_img_path)}")
plt.show()

# 打印检测结果
for r in results:
    for box in r.boxes:
        cls = int(box.cls[0])
        conf = float(box.conf[0])
        label = yaml_data['names'][cls]
        print(f"🔍 检测到: {label} | 置信度: {conf:.4f}")
